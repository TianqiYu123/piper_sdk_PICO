sudo nano /etc/systemd/system/piper_service.service

Paste the following content into the file, modifying the paths to match your setup:

[Unit]
Description=Run piper_web.py on startup
After=final.target
Wants=final.target

[Service]
Environment="PATH=/home/tq/.local/bin:/home/tq/bin:/home/tq/miniforge3/bin:/hom>
WorkingDirectory=/home/tq/robotarm/piper_sdk_PICO
ExecStart= /home/tq/robotarm/piper_sdk_PICO/piper.sh
Restart=on-failure
RestartSec=100

[Install]
WantedBy=final.target


sudo systemctl enable piper_service.service
sudo systemctl start piper_service.service
sudo systemctl status piper_service.service
sudo journalctl -u piper_service.service
